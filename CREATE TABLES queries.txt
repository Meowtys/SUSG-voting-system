CREATE TABLE Colleges (
    college_id INT(4) AUTO_INCREMENT PRIMARY KEY,
    college_name VARCHAR(255) NOT NULL
    max_representatives NOT NULL, DEFAULT 1
);

CREATE TABLE Positions (
    position_id INT(4) AUTO_INCREMENT PRIMARY KEY,
    position_name VARCHAR(255) NOT NULL
);

CREATE TABLE Students (
    student_id INT(4) AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(255) NOT NULL,
    stud_name VARCHAR(255) NOT NULL,
    college_id INT(4),
    is_admin BOOLEAN NOT NULL,
    password VARCHAR(255) NOT NULL,
    has_voted BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (college_id) REFERENCES Colleges(college_id)
);

CREATE TABLE Candidates (
    candidate_id INT(4) AUTO_INCREMENT PRIMARY KEY,
    student_id INT(4),
    college_id INT(4),
    position_id INT(4),
    qualified BOOLEAN DEFAULT FALSE,
    remarks VARCHAR(255),
    FOREIGN KEY (student_id) REFERENCES Students(student_id),
    FOREIGN KEY (college_id) REFERENCES Colleges(college_id),
    FOREIGN KEY (position_id) REFERENCES Positions(position_id)
);

CREATE TABLE Votes (
    vote_id INT(4) AUTO_INCREMENT PRIMARY KEY,
    student_id INT(4),
    candidate_id INT(4),
    position_id INT(4),
    vote_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (student_id) REFERENCES Students(student_id),
    FOREIGN KEY (candidate_id) REFERENCES Candidates(candidate_id),
    FOREIGN KEY (position_id) REFERENCES Positions(position_id)
);

CREATE TABLE Feedback (
    feedback_id INT(4) AUTO_INCREMENT PRIMARY KEY,
    student_id INT(4),
    comment TEXT NOT NULL,
    feedback_date DATETIME NOT NULL,
    FOREIGN KEY (student_id) REFERENCES Students(student_id) ON DELETE CASCADE ON UPDATE CASCADE
);
